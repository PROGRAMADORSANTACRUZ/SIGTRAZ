    PUNTOS_VENTA ||--o{ DEVOLUCIONES : "punto_venta_id"
    PUNTOS_VENTA ||--o{ DOCUMENTALES : "punto_venta_id"
    PRODUCTOS ||--o{ ENTRADAS : "producto_id"
    PRODUCTOS ||--o{ ACONDICIONAMIENTO : "producto_id"
    PRODUCTOS ||--o{ SALIDAS : "producto_id"
    FICHAS_TECNICAS ||--o{ ACONDICIONAMIENTO : "vida util"
    USUARIOS {
        int id PK
        string email
        string rol
        string password_hash
    }
    PUNTOS_VENTA {
        int id PK
        string pdv
        string prefijo
    }
    ENTRADAS {
        int id PK
        string lote_interno
        string producto_id FK
        int punto_venta_id FK
        bool editado
    }
    DOCUMENTALES {
        string nota "vehiculo, POES, LYD, higiene,"
        string nota2 "agua, temperatura (10 tablas)"
    }

DOCUMENTALES representa las 10 tablas de Gestion documental que comparten la columna punto_venta_id. Los catalogos (productos, proveedores, fichas, puntos_venta) NO estan segmentados por PDV.
